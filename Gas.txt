// TỐI ƯU TỐC ĐỘ XỬ LÝ (SUB-SECOND), HỖ TRỢ XUẤT SHEET THU TẠM & FILE JSON NHÁP
// TÍCH HỢP ĐẦY ĐỦ: NGÂN HÀNG CÂU HỎI, KHÔI PHỤC BACKUP, LƯU & TẢI LẠI ẢNH 2 TẦNG

const ROOT_BACKUP_FOLDER_NAME = "FILE_SHEET_AND_JSON_DU_PHONG";
const ROOT_TULUAN_FOLDER_NAME = "BAI_TAP_TU_LUAN";
const ROOT_DOCS_FOLDER_NAME   = "KHO_TAI_LIEU_PDF_WORD_ANH_VIDEO";

const ALLOWED_8_CLASSES = [
  "Thêm 10",
  "Thêm 11",
  "Thêm 12",
  "Lớp 11A",
  "Lớp 11C",
  "Lớp 11E",
  "Lớp 10P",
  "HSG Toán 11"
];

// =============================================================================
// 1. KIỂM TRA TRẠNG THÁI MÁY CHỦ (GET)
// =============================================================================
function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({
    status: "online",
    message: "Máy chủ Google Apps Script Drive 5TB đang vận hành chuẩn hóa 8 lớp, hỗ trợ Ngân hàng câu hỏi & Ảnh 2 tầng!",
    rootFolder: ROOT_BACKUP_FOLDER_NAME,
    allowedClasses: ALLOWED_8_CLASSES,
    timestamp: new Date().toISOString()
  })).setMimeType(ContentService.MimeType.JSON);
}

// =============================================================================
// 2. TIẾP NHẬN YÊU CẦU NỘP BÀI, SAO LƯU VÀ QUẢN LÝ TỆP (POST)
// =============================================================================
function doPost(e) {
  var lock = LockService.getScriptLock();
  var hasLock = false;

  try {
    hasLock = lock.tryLock(45000);
  } catch (eLock) {
    hasLock = false;
  }

  if (!hasLock) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "busy",
      message: "Máy chủ đang xử lý lượt nộp khác, hệ thống sẽ tự động thử lại ngay..."
    })).setMimeType(ContentService.MimeType.JSON);
  }

  try {
    if (!e || !e.postData || !e.postData.contents) {
      throw new Error("Không nhận được dữ liệu tải lên.");
    }

    var rawData = e.postData.contents;
    var data = JSON.parse(rawData);
    var action = data.action || "";

    // -------------------------------------------------------------------------
    // NHIỆM VỤ 1: NỘP BÀI CHÍNH THỨC (XUẤT FILE JSON GHI ĐÈ & GHI SHEET ĐIỂM CUỐI)
    // -------------------------------------------------------------------------
    if (action === "saveSubmissionJsonFile" || action === "backupSubmissionRecord") {
      var sub = data.submission || {};
      var sbd = String(sub.sbd || data.sbd || "").trim();
      var studentName = String(sub.student_name || data.student_name || sub.name || data.name || "").trim();

      if (!sbd && !studentName) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "ignored",
          message: "Bỏ qua: Không có thông tin thí sinh."
        })).setMimeType(ContentService.MimeType.JSON);
      }

      var rawClass = data.className || data.class || data.lop || sub.class_name || sub.className || sub.studentClass || sub.category_id || "";
      var className = standardizeClassName(rawClass, sbd);

      var rawExamTitle = data.examTitle || data.folderKey || data.title || sub.exam_title || sub.examTitle || sub.title || "";
      var examTitle = sanitizeExamTitle(rawExamTitle);

      if (!examTitle || examTitle === "De_Thi" || examTitle === "Bai_Tap") {
        return ContentService.createTextOutput(JSON.stringify({
          status: "ignored",
          message: "Bỏ qua: Thiếu tên đề thi thực tế."
        })).setMimeType(ContentService.MimeType.JSON);
      }

      var rootFolder = getCachedFolderExact(DriveApp.getRootFolder(), ROOT_BACKUP_FOLDER_NAME, "ROOT_BACKUP");
      var classFolder = getCachedFolderExact(rootFolder, className, "CLASS_" + className);
      var examFolder = getCachedFolderExact(classFolder, examTitle, "EXAM_" + className + "_" + examTitle);

      var jsonFileName = cleanFilePartName(sbd) + "_" + cleanFilePartName(studentName) + ".json";
      var jsonContent = "";
      if (data.base64) {
        var cleanB64 = String(data.base64).replace(/\s+/g, '');
        if (cleanB64.indexOf("base64,") !== -1) cleanB64 = cleanB64.split("base64,")[1];
        var decodedBytes = Utilities.base64Decode(cleanB64, Utilities.Charset.UTF_8);
        jsonContent = Utilities.newBlob(decodedBytes, "application/json").getDataAsString("UTF-8");
      } else {
        jsonContent = JSON.stringify(sub, null, 2);
      }

      var savedJsonFile = saveOrOverwriteJsonFile(examFolder, jsonFileName, jsonContent);

      try {
        var finalSheetName = "file điểm thi cuối_" + examTitle + "_sheet";
        recordSubmissionToTargetSheet(examFolder, finalSheetName, className, sub, false);
      } catch (sheetErr) {
        Logger.log("Lỗi ghi Sheet cuối: " + sheetErr);
      }

      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        message: "Đã lưu đè JSON mới nhất và ghi file điểm thi cuối thành công!",
        className: className,
        folderUrl: examFolder.getUrl(),
        fileUrl: savedJsonFile ? savedJsonFile.getUrl() : ""
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // -------------------------------------------------------------------------
    // NHIỆM VỤ 2: KHI BẤM NÚT "THU TẠM" (XUẤT SHEET THU TẠM & LƯU FILE JSON NHÁP)
    // -------------------------------------------------------------------------
    if (action === "saveLiveDraftToSheet" || action === "autoSaveDraftRecord") {
      var draft = data.draft || data.submission || {};
      var sbdDraft = String(draft.sbd || data.sbd || "").trim();
      var studentNameDraft = String(draft.student_name || data.student_name || draft.name || data.name || "").trim();

      if (!sbdDraft && !studentNameDraft) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "ignored",
          message: "Bỏ qua: Bản nháp thiếu thông tin thí sinh."
        })).setMimeType(ContentService.MimeType.JSON);
      }

      var rawClassDraft = data.className || data.class || draft.class_name || draft.className || draft.category_id || "";
      var classDraftName = standardizeClassName(rawClassDraft, sbdDraft);

      var rawExamDraft = data.examTitle || data.title || draft.exam_title || draft.examTitle || "";
      var examDraftTitle = sanitizeExamTitle(rawExamDraft);

      if (!examDraftTitle) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "ignored",
          message: "Bỏ qua: Thiếu tên đề thi cho bản nháp."
        })).setMimeType(ContentService.MimeType.JSON);
      }

      var rootF = getCachedFolderExact(DriveApp.getRootFolder(), ROOT_BACKUP_FOLDER_NAME, "ROOT_BACKUP");
      var classF = getCachedFolderExact(rootF, classDraftName, "CLASS_" + classDraftName);
      var examF = getCachedFolderExact(classF, examDraftTitle, "EXAM_" + classDraftName + "_" + examDraftTitle);

      var savedDraftFileUrl = "";
      try {
        var draftJsonName = "tam_" + cleanFilePartName(sbdDraft) + "_" + cleanFilePartName(studentNameDraft) + ".json";
        var draftContent = JSON.stringify(draft, null, 2);
        var draftFile = saveOrOverwriteJsonFile(examF, draftJsonName, draftContent);
        if (draftFile) savedDraftFileUrl = draftFile.getUrl();
      } catch(eJson) {}

      var tempSheetName = "file điểm thi thu tạm_" + examDraftTitle + "_sheet";
      recordSubmissionToTargetSheet(examF, tempSheetName, classDraftName, draft, true);

      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        className: classDraftName,
        message: "Đã xuất kết quả thu tạm vào Sheet và lưu file JSON nháp thành công!",
        fileUrl: savedDraftFileUrl
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // -------------------------------------------------------------------------
    // NHIỆM VỤ 3: TẠO SẴN THƯ MỤC ĐỀ THI
    // -------------------------------------------------------------------------
    if (action === "createExamFolders") {
      var rawClassC = data.className || data.class || data.lop || "";
      var rawExamC = data.examTitle || data.folderKey || data.title || "";
      var classFolderC = standardizeClassName(rawClassC, data.sbd || "");
      var examTitleC = sanitizeExamTitle(rawExamC);

      if (!classFolderC || !examTitleC) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "ignored",
          message: "Bỏ qua: Thiếu tên lớp hoặc tên đề thi hợp lệ."
        })).setMimeType(ContentService.MimeType.JSON);
      }

      var rootFolderD = getCachedFolderExact(DriveApp.getRootFolder(), ROOT_BACKUP_FOLDER_NAME, "ROOT_BACKUP");
      var classFolderD = getCachedFolderExact(rootFolderD, classFolderC, "CLASS_" + classFolderC);
      var examFolderD = getCachedFolderExact(classFolderD, examTitleC, "EXAM_" + classFolderC + "_" + examTitleC);

      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        className: classFolderC,
        folderUrl: examFolderD.getUrl()
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // -------------------------------------------------------------------------
    // NHIỆM VỤ 4: TẢI LÊN FILE ĐÁP ÁN CHẤM TỰ LUẬN
    // -------------------------------------------------------------------------
    if (action === "uploadDapAnCham") {
      var rootFolderDoc = getOrCreateFolderExact(DriveApp.getRootFolder(), ROOT_DOCS_FOLDER_NAME);
      var targetFolder = getOrCreateFolderExact(rootFolderDoc, "Dap_an_cham_tu_luan");

      var fileName = sanitizeExamTitle(data.filename || ("DapAn_" + Date.now()));
      var mimeType = data.mimeType || "application/octet-stream";
      var rawB64 = String(data.base64 || "").replace(/\s+/g, '');
      if (rawB64.indexOf("base64,") !== -1) rawB64 = rawB64.split("base64,")[1];
      var decodedBytes = Utilities.base64Decode(rawB64);

      deleteFilesMatchingName(targetFolder, fileName);

      var file = targetFolder.createFile(Utilities.newBlob(decodedBytes, mimeType, fileName));
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        fileUrl: file.getUrl(),
        fileId: file.getId(),
        directUrl: "https://lh3.googleusercontent.com/d/" + file.getId()
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // -------------------------------------------------------------------------
    // NHIỆM VỤ 5: TẢI ẢNH BÀI LÀM TỰ LUẬN CỦA HỌC SINH
    // -------------------------------------------------------------------------
    if (data.studentFolder && data.base64) {
      var rootFolderTL = getOrCreateFolderExact(DriveApp.getRootFolder(), ROOT_TULUAN_FOLDER_NAME);
      var classTLName = standardizeClassName(data.className || data.class || data.lop || "lop-11a", data.sbd || "");
      var examTLName = sanitizeExamTitle(data.examTitle || data.folderKey || "Bai_Tap");

      var classFolderTL = getOrCreateFolderExact(rootFolderTL, classTLName);
      var examFolderTL = getOrCreateFolderExact(classFolderTL, examTLName);
      var studentFolderTL = getOrCreateFolderExact(examFolderTL, sanitizeExamTitle(data.studentFolder));

      var fileNameTL = sanitizeExamTitle(data.filename || ("Trang_" + Date.now() + ".jpg"));
      deleteFilesMatchingName(studentFolderTL, fileNameTL);

      var cleanB64TL = String(data.base64).replace(/\s+/g, '');
      if (cleanB64TL.indexOf("base64,") !== -1) cleanB64TL = cleanB64TL.split("base64,")[1];
      var decodedBytesTL = Utilities.base64Decode(cleanB64TL);
      var blobTL = Utilities.newBlob(decodedBytesTL, data.mimeType || "image/jpeg", fileNameTL);
      var fileTL = studentFolderTL.createFile(blobTL);
      fileTL.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        className: classTLName,
        fileUrl: fileTL.getUrl(),
        fileId: fileTL.getId(),
        directUrl: "https://lh3.googleusercontent.com/d/" + fileTL.getId()
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // -------------------------------------------------------------------------
    // NHIỆM VỤ 8: LƯU TRỮ NGÂN HÀNG CÂU HỎI & WORKSPACE BACKUP LÊN DRIVE
    // -------------------------------------------------------------------------
    if (action === "saveQuestionBankData") {
      try {
        var pathString = data.path || (ROOT_DOCS_FOLDER_NAME + "/Kho_ngan_hang_cau_hoi/Workspace_Backup");
        var filename = data.filename || "NganHangCauHoi_Workspace.json";
        var currentFolder = getCachedPathFolder(pathString);
        var jsonContentQB = JSON.stringify(data.data, null, 2);
        var savedFileQB = saveOrOverwriteJsonFile(currentFolder, filename, jsonContentQB);
        if (savedFileQB) {
          savedFileQB.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        }

        return ContentService.createTextOutput(JSON.stringify({
          status: "success",
          fileUrl: savedFileQB ? savedFileQB.getUrl() : "",
          folderUrl: currentFolder.getUrl(),
          message: "Lưu Ngân hàng câu hỏi vào Google Drive thành công!"
        })).setMimeType(ContentService.MimeType.JSON);
      } catch (errQB) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error",
          message: "Lỗi tạo thư mục/file GAS (Question Bank): " + errQB.toString()
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // -------------------------------------------------------------------------
    // NHIỆM VỤ 9: ĐỌC FILE WORKSPACE BACKUP TỪ DRIVE ĐỂ KHÔI PHỤC NGÂN HÀNG ĐỀ
    // -------------------------------------------------------------------------
    if (action === "loadQuestionBankData") {
      try {
        var loadPath = data.path || (ROOT_DOCS_FOLDER_NAME + "/Kho_ngan_hang_cau_hoi/Workspace_Backup");
        var loadFilename = data.filename || "NganHangCauHoi_Workspace.json";
        var targetLoadFolder = findPathFolderReadOnly(loadPath);
        if (!targetLoadFolder) {
          return ContentService.createTextOutput(JSON.stringify({
            status: "error",
            message: "Không tìm thấy thư mục Backup: " + loadPath
          })).setMimeType(ContentService.MimeType.JSON);
        }

        var loadFiles = targetLoadFolder.getFilesByName(loadFilename);
        if (!loadFiles.hasNext()) {
          return ContentService.createTextOutput(JSON.stringify({
            status: "error",
            message: "Không tìm thấy tệp " + loadFilename + " trong thư mục Backup trên Drive."
          })).setMimeType(ContentService.MimeType.JSON);
        }

        var backupFile = loadFiles.next();
        var rawJsonStr = backupFile.getBlob().getDataAsString("UTF-8");
        var parsedBackup = JSON.parse(rawJsonStr);

        return ContentService.createTextOutput(JSON.stringify({
          status: "success",
          data: parsedBackup,
          fileUrl: backupFile.getUrl(),
          folderUrl: targetLoadFolder.getUrl()
        })).setMimeType(ContentService.MimeType.JSON);
      } catch (errLoad) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error",
          message: "Lỗi đọc file Backup từ Drive: " + errLoad.toString()
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // -------------------------------------------------------------------------
    // NHIỆM VỤ 10: LƯU ẢNH DỰ PHÒNG 2 TẦNG CHO TỪNG CÂU HỎI VÀO ĐÚNG THƯ MỤC
    // -------------------------------------------------------------------------
    if (action === "uploadQuestionImageBackup") {
      var imgPath = data.path || data.folderName || (ROOT_DOCS_FOLDER_NAME + "/Kho_anh_ngan_hang_cau_hoi/Chung");
      var imgFolder = getCachedPathFolder(imgPath);
      var imgFileName = sanitizeFileNameKeepExt(data.filename || ("Cau_Hoi_" + Date.now() + ".png"));
      var imgMime = data.mimeType || "image/png";
      var imgB64 = String(data.base64 || "").replace(/\s+/g, '');
      if (imgB64.indexOf("base64,") !== -1) imgB64 = imgB64.split("base64,")[1];

      var imgBytes = Utilities.base64Decode(imgB64);
      deleteFilesMatchingName(imgFolder, imgFileName);

      var savedImgFile = imgFolder.createFile(Utilities.newBlob(imgBytes, imgMime, imgFileName));
      if (data.description) {
        try { savedImgFile.setDescription(String(data.description)); } catch (descErr) {}
      }
      savedImgFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        fileId: savedImgFile.getId(),
        fileUrl: savedImgFile.getUrl(),
        directUrl: "https://lh3.googleusercontent.com/d/" + savedImgFile.getId(),
        folderUrl: imgFolder.getUrl(),
        folderPath: imgPath,
        filename: imgFileName
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // -------------------------------------------------------------------------
    // NHIỆM VỤ 12: TẢI ẢNH TỪ URL NGOÀI (IMGBB/WEB) LƯU THẲNG VÀO DRIVE (CHỐNG CORS)
    // -------------------------------------------------------------------------
    if (action === "backupImageUrlToDrive") {
      try {
        var sourceUrl = String(data.imageUrl || "").trim();
        if (!sourceUrl) throw new Error("Thiếu đường dẫn ảnh imageUrl.");

        var resp = UrlFetchApp.fetch(sourceUrl, {
          muteHttpExceptions: true,
          followRedirects: true
        });
        var code = resp.getResponseCode();
        if (code < 200 || code >= 300) {
          throw new Error("Link ảnh nguồn không còn khả dụng (HTTP " + code + ").");
        }

        var blobSrc = resp.getBlob();
        var mimeSrc = String(blobSrc.getContentType() || "").toLowerCase();
        var bytesSrc = blobSrc.getBytes();
        if (!bytesSrc || bytesSrc.length < 100 || mimeSrc.indexOf("text/html") !== -1) {
          throw new Error("Tệp trả về từ URL không phải ảnh hợp lệ.");
        }

        var extByMime = "png";
        if (mimeSrc.indexOf("jpeg") !== -1 || mimeSrc.indexOf("jpg") !== -1) extByMime = "jpg";
        else if (mimeSrc.indexOf("webp") !== -1) extByMime = "webp";
        else if (mimeSrc.indexOf("gif") !== -1) extByMime = "gif";

        var urlPath = data.path || data.folderName || (ROOT_DOCS_FOLDER_NAME + "/Kho_anh_ngan_hang_cau_hoi/Chung");
        var urlFolder = getCachedPathFolder(urlPath);
        var rawUrlFname = sanitizeFileNameKeepExt(data.filename || ("Cau_Hoi_" + Date.now() + "." + extByMime));
        var urlFileName = rawUrlFname.replace(/\.(png|jpg|jpeg|webp|gif)$/i, "." + extByMime);

        deleteFilesMatchingName(urlFolder, urlFileName);

        var finalBlob = Utilities.newBlob(bytesSrc, mimeSrc.indexOf("image/") === 0 ? mimeSrc : ("image/" + extByMime), urlFileName);
        var savedUrlFile = urlFolder.createFile(finalBlob);
        if (data.description) {
          try { savedUrlFile.setDescription(String(data.description)); } catch (descUrlErr) {}
        }
        savedUrlFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

        return ContentService.createTextOutput(JSON.stringify({
          status: "success",
          fileId: savedUrlFile.getId(),
          fileUrl: savedUrlFile.getUrl(),
          directUrl: "https://lh3.googleusercontent.com/d/" + savedUrlFile.getId(),
          folderUrl: urlFolder.getUrl(),
          folderPath: urlPath,
          filename: urlFileName
        })).setMimeType(ContentService.MimeType.JSON);
      } catch (errUrlBk) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error",
          message: errUrlBk.toString()
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // -------------------------------------------------------------------------
    // NHIỆM VỤ 11: TRUY XUẤT TRỰC TIẾP ẢNH TỪ DRIVE TRẢ VỀ BASE64 (CHO NÚT 🔄 TẢI LẠI ẢNH)
    // -------------------------------------------------------------------------
    if (action === "fetchDriveImageBase64") {
      try {
        var targetImgFile = null;
        var parentFolderUrl = "";

        if (data.fileId) {
          try {
            var candidate = DriveApp.getFileById(String(data.fileId).trim());
            if (candidate && !candidate.isTrashed()) targetImgFile = candidate;
          } catch (fIdErr) {}
        }

        if (!targetImgFile && data.path && data.filename) {
          var searchFolder = findPathFolderReadOnly(data.path);
          if (searchFolder) {
            parentFolderUrl = searchFolder.getUrl();
            var matchFiles = searchFolder.getFilesByName(String(data.filename).trim());
            if (matchFiles.hasNext()) targetImgFile = matchFiles.next();
          }
        }

        if (!targetImgFile) {
          return ContentService.createTextOutput(JSON.stringify({
            status: "error",
            message: "Không tìm thấy tệp ảnh dự phòng trên Google Drive."
          })).setMimeType(ContentService.MimeType.JSON);
        }

        var imgBlob = targetImgFile.getBlob();
        var contentType = imgBlob.getContentType() || "image/png";
        var encodedB64 = Utilities.base64Encode(imgBlob.getBytes());
        var dataUrl = "data:" + contentType + ";base64," + encodedB64;

        if (!parentFolderUrl) {
          try {
            var parents = targetImgFile.getParents();
            if (parents.hasNext()) parentFolderUrl = parents.next().getUrl();
          } catch (pErr) {}
        }

        return ContentService.createTextOutput(JSON.stringify({
          status: "success",
          dataUrl: dataUrl,
          directUrl: "https://lh3.googleusercontent.com/d/" + targetImgFile.getId(),
          fileId: targetImgFile.getId(),
          fileUrl: targetImgFile.getUrl(),
          folderUrl: parentFolderUrl
        })).setMimeType(ContentService.MimeType.JSON);
      } catch (errFetchImg) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error",
          message: "Lỗi đọc ảnh từ Drive: " + errFetchImg.toString()
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // -------------------------------------------------------------------------
    // NHIỆM VỤ 6: TẢI TỆP TÀI LIỆU CHUNG (PDF / WORD / EXCEL / ẢNH) VÀO KHO
    // -------------------------------------------------------------------------
    if (action === "uploadFile" || action === "uploadDocument" || (data.base64 && data.filename && !data.studentFolder)) {
      var rootFolderDocs = getOrCreateFolderExact(DriveApp.getRootFolder(), ROOT_DOCS_FOLDER_NAME);
      var targetFolderDoc = rootFolderDocs;
      var rawFolderTarget = data.folderName || data.className || data.class || "";
      if (rawFolderTarget) {
        targetFolderDoc = getOrCreateFolderExact(rootFolderDocs, sanitizeExamTitle(rawFolderTarget));
      }

      var docFileName = sanitizeExamTitle(data.filename || ("Tep_" + Date.now()));
      var docMimeType = data.mimeType || "application/octet-stream";
      var rawB64Doc = String(data.base64 || "").replace(/\s+/g, '');
      if (rawB64Doc.indexOf("base64,") !== -1) rawB64Doc = rawB64Doc.split("base64,")[1];
      var decodedDocBytes = Utilities.base64Decode(rawB64Doc);

      var newDocFile = targetFolderDoc.createFile(Utilities.newBlob(decodedDocBytes, docMimeType, docFileName));
      newDocFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        fileUrl: newDocFile.getUrl(),
        fileId: newDocFile.getId(),
        directUrl: "https://lh3.googleusercontent.com/d/" + newDocFile.getId()
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // -------------------------------------------------------------------------
    // NHIỆM VỤ 7: XÓA FILE THEO ID
    // -------------------------------------------------------------------------
    if (action === "deleteFile" && data.fileId) {
      try {
        DriveApp.getFileById(data.fileId).setTrashed(true);
        return ContentService.createTextOutput(JSON.stringify({
          status: "success",
          message: "Đã xóa file thành công."
        })).setMimeType(ContentService.MimeType.JSON);
      } catch (delErr) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "ignored",
          message: delErr.toString()
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    return ContentService.createTextOutput(JSON.stringify({
      status: "ignored",
      message: "Không có tác vụ phù hợp."
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    try {
      lock.releaseLock();
    } catch (eLock) {}
  }
}

function computeCacheHashKey(prefix, rawInput) {
  var bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, String(rawInput || ""), Utilities.Charset.UTF_8);
  var b64 = Utilities.base64EncodeWebSafe(bytes).replace(/[^a-zA-Z0-9_]/g, "");
  return String(prefix || "K_") + b64;
}

function standardizeClassName(rawClass, sbd) {
  var sbdClean = String(sbd || "").toLowerCase().trim();
  var classStr = String(rawClass || "").toLowerCase().trim();

  if (/^11\d{2}$/.test(sbdClean)) return "Thêm 11";
  if (/^10\d{2}$/.test(sbdClean)) return "Thêm 10";
  if (/^12\d{2}$/.test(sbdClean)) return "Thêm 12";

  if (sbdClean.indexOf("hsg") === 0) return "HSG Toán 11";
  if (sbdClean.indexOf("11a") === 0) return "Lớp 11A";
  if (sbdClean.indexOf("11c") === 0) return "Lớp 11C";
  if (sbdClean.indexOf("11e") === 0) return "Lớp 11E";
  if (sbdClean.indexOf("10p") === 0) return "Lớp 10P";

  if (classStr.indexOf("hsg") !== -1) return "HSG Toán 11";
  if (classStr.indexOf("them-11") !== -1 || classStr.indexOf("thêm 11") !== -1 || classStr.indexOf("them 11") !== -1) return "Thêm 11";
  if (classStr.indexOf("them-10") !== -1 || classStr.indexOf("thêm 10") !== -1 || classStr.indexOf("them 10") !== -1) return "Thêm 10";
  if (classStr.indexOf("them-12") !== -1 || classStr.indexOf("thêm 12") !== -1 || classStr.indexOf("them 12") !== -1) return "Thêm 12";

  if (classStr.indexOf("11a") !== -1) return "Lớp 11A";
  if (classStr.indexOf("11c") !== -1) return "Lớp 11C";
  if (classStr.indexOf("11e") !== -1) return "Lớp 11E";
  if (classStr.indexOf("10p") !== -1) return "Lớp 10P";

  return "Thêm 11";
}

function getCachedFolderExact(parentFolder, folderName, cacheKey) {
  var cache = CacheService.getScriptCache();
  var cleanKey = computeCacheHashKey("FLD2_", parentFolder.getId() + "|" + String(folderName || "") + "|" + String(cacheKey || ""));
  var cachedId = cache.get(cleanKey);

  if (cachedId) {
    try {
      var folder = DriveApp.getFolderById(cachedId);
      if (folder && !folder.isTrashed()) return folder;
    } catch (eCache) {}
  }

  var folderExact = getOrCreateFolderExact(parentFolder, folderName);
  try {
    cache.put(cleanKey, folderExact.getId(), 21600);
  } catch (ePut) {}

  return folderExact;
}

function getCachedPathFolder(pathString) {
  var normPath = String(pathString || "").trim();
  var cache = CacheService.getScriptCache();
  var cacheKey = computeCacheHashKey("PTH2_", normPath);
  var cachedId = cache.get(cacheKey);

  if (cachedId) {
    try {
      var folder = DriveApp.getFolderById(cachedId);
      if (folder && !folder.isTrashed()) return folder;
    } catch (eCache) {}
  }

  var created = getOrCreatePathFolder(normPath);
  try {
    cache.put(cacheKey, created.getId(), 21600);
  } catch (ePut) {}
  return created;
}

function getOrCreateFolderExact(parentFolder, exactFolderName) {
  var name = String(exactFolderName || "").trim();
  var folders = parentFolder.getFoldersByName(name);
  if (folders.hasNext()) {
    return folders.next();
  }
  return parentFolder.createFolder(name);
}

function getOrCreatePathFolder(pathString) {
  var parts = String(pathString || "").split("/");
  var cur = DriveApp.getRootFolder();
  for (var i = 0; i < parts.length; i++) {
    var seg = String(parts[i] || "").trim();
    if (!seg) continue;
    cur = getOrCreateFolderExact(cur, seg);
  }
  return cur;
}

function findPathFolderReadOnly(pathString) {
  var parts = String(pathString || "").split("/");
  var cur = DriveApp.getRootFolder();
  for (var i = 0; i < parts.length; i++) {
    var seg = String(parts[i] || "").trim();
    if (!seg) continue;
    var folders = cur.getFoldersByName(seg);
    if (!folders.hasNext()) return null;
    cur = folders.next();
  }
  return cur;
}

function saveOrOverwriteJsonFile(folder, fileName, jsonString) {
  var files = folder.getFilesByName(fileName);
  var targetFile = null;

  if (files.hasNext()) {
    targetFile = files.next();
    targetFile.setContent(jsonString);
    while (files.hasNext()) {
      try { files.next().setTrashed(true); } catch (eT) {}
    }
  } else {
    targetFile = folder.createFile(fileName, jsonString, "application/json");
  }

  try {
    var baseName = fileName.replace(/\.json$/i, "");
    var dup1 = folder.getFilesByName(baseName + " (1).json");
    while (dup1.hasNext()) { try { dup1.next().setTrashed(true); } catch(e1) {} }
    var dup2 = folder.getFilesByName(baseName + " (2).json");
    while (dup2.hasNext()) { try { dup2.next().setTrashed(true); } catch(e2) {} }
  } catch(eClean) {}

  return targetFile;
}

function deleteFilesMatchingName(folder, fileName) {
  var files = folder.getFilesByName(fileName);
  while (files.hasNext()) {
    try { files.next().setTrashed(true); } catch (eDel) {}
  }
}

function recordSubmissionToTargetSheet(folder, sheetTitle, className, sub, isTempDraft) {
  if (!sub || (!sub.sbd && !sub.student_name)) return;

  var cache = CacheService.getScriptCache();
  var sheetCacheKey = computeCacheHashKey("SHT2_", folder.getId() + "|" + (isTempDraft ? "TEMP" : "FINAL"));
  var cachedSheetId = cache.get(sheetCacheKey);
  var spreadsheet = null;

  if (cachedSheetId) {
    try {
      spreadsheet = SpreadsheetApp.openById(cachedSheetId);
    } catch (eOpen) {}
  }

  if (!spreadsheet) {
    var files = folder.getFilesByName(sheetTitle);
    if (files.hasNext()) {
      spreadsheet = SpreadsheetApp.open(files.next());
    } else {
      spreadsheet = SpreadsheetApp.create(sheetTitle);
      var sheetFile = DriveApp.getFileById(spreadsheet.getId());
      sheetFile.moveTo(folder);

      var sheet = spreadsheet.getActiveSheet();
      sheet.setName("KetQua");

      var headers = isTempDraft ? [
        "Thời gian cập nhật nháp",
        "Lớp",
        "Số báo danh",
        "Họ và tên học sinh",
        "Điểm tạm tính",
        "Số câu đã chọn",
        "Thời gian làm",
        "Số lần chuyển tab",
        "Chi tiết phương án đã chọn"
      ] : [
        "Thời gian nộp chính thức",
        "Lớp",
        "Số báo danh",
        "Họ và tên học sinh",
        "Điểm thang 10",
        "Điểm gốc",
        "Số câu đúng",
        "Số câu sai",
        "Thời gian làm",
        "Số lần chuyển tab",
        "Chi tiết phương án đã chọn"
      ];

      sheet.appendRow(headers);
      var headerRange = sheet.getRange(1, 1, 1, headers.length);
      headerRange.setBackground(isTempDraft ? "#0d9488" : "#0284c7");
      headerRange.setFontColor("#ffffff");
      headerRange.setFontWeight("bold");
      headerRange.setHorizontalAlignment("center");
      sheet.setFrozenRows(1);
    }

    try {
      cache.put(sheetCacheKey, spreadsheet.getId(), 21600);
    } catch (eC) {}
  }

  var activeSheet = spreadsheet.getSheetByName("KetQua") || spreadsheet.getActiveSheet();
  var nowStr = Utilities.formatDate(new Date(), "GMT+7", "HH:mm:ss dd/MM/yyyy");
  var sbd = String(sub.sbd || "").trim();
  var name = String(sub.student_name || sub.name || "").trim();
  var compTime = sub.completion_time || (sub.spent_mins ? (sub.spent_mins + " phút") : "");
  var tabSwitch = sub.tab_switch_count || "0 lần";

  var dataStr = sub.data_string || "";
  if (!dataStr && sub.user_answers && typeof sub.user_answers === "object") {
    var arrAns = [];
    for (var k in sub.user_answers) {
      arrAns.push(k + ":" + sub.user_answers[k]);
    }
    dataStr = arrAns.join(" | ");
  }

  var rowData = [];
  if (isTempDraft) {
    var answeredCount = 0;
    if (sub.user_answers && typeof sub.user_answers === "object") {
      answeredCount = Object.keys(sub.user_answers).length;
    }

    var tempScore = "Đang làm...";
    if (sub.score10 !== undefined && sub.score10 !== null && sub.score10 !== "") {
      var correctTxt = (sub.correct_count !== undefined) ? (" (" + sub.correct_count + " đúng)") : "";
      tempScore = sub.score10 + " đ" + correctTxt;
    }

    rowData = [
      nowStr,
      className,
      sbd,
      name,
      tempScore,
      answeredCount + " câu",
      compTime,
      tabSwitch,
      dataStr
    ];
  } else {
    var score10 = (sub.score10 !== undefined) ? sub.score10 : "";
    var rawScore = (sub.raw_score !== undefined) ? sub.raw_score : "";
    var correct = (sub.correct_count !== undefined) ? sub.correct_count : "";
    var wrong = (sub.wrong_count !== undefined) ? sub.wrong_count : "";
    rowData = [
      nowStr,
      className,
      sbd,
      name,
      score10,
      rawScore,
      correct,
      wrong,
      compTime,
      tabSwitch,
      dataStr
    ];
  }

  var lastRow = activeSheet.getLastRow();
  var targetRowIndex = -1;

  if (lastRow > 1) {
    var sbdColumnValues = activeSheet.getRange(2, 3, lastRow - 1, 1).getValues();
    for (var i = 0; i < sbdColumnValues.length; i++) {
      var existingSbd = String(sbdColumnValues[i][0]).trim().toLowerCase();
      if (existingSbd === sbd.toLowerCase()) {
        targetRowIndex = i + 2;
        break;
      }
    }
  }

  if (targetRowIndex !== -1) {
    activeSheet.getRange(targetRowIndex, 1, 1, rowData.length).setValues([rowData]);
  } else {
    activeSheet.appendRow(rowData);
  }

  SpreadsheetApp.flush();
}

function manualTriggerFullDriveCleanup() {
  Logger.log("🚀 Bắt đầu quét và gộp thư mục trên Drive...");
  var rootFolders = DriveApp.getFoldersByName(ROOT_BACKUP_FOLDER_NAME);
  while (rootFolders.hasNext()) {
    var rf = rootFolders.next();
    var subF = rf.getFolders();
    var list = [];
    while (subF.hasNext()) list.push(subF.next());
    for (var i = 0; i < list.length; i++) {
      var curF = list[i];
      var stdName = standardizeClassName(curF.getName(), "");
      if (curF.getName().trim() !== stdName) {
        var targetF = getOrCreateFolderExact(rf, stdName);
        var inF = curF.getFolders();
        while (inF.hasNext()) {
          var inf = inF.next();
          try {
            var destSub = getOrCreateFolderExact(targetF, inf.getName());
            var infFiles = inf.getFiles();
            while (infFiles.hasNext()) infFiles.next().moveTo(destSub);
            inf.setTrashed(true);
          } catch(e) {}
        }
        var inFiles = curF.getFiles();
        while (inFiles.hasNext()) inFiles.next().moveTo(targetF);
        curF.setTrashed(true);
      }
    }
  }
  Logger.log("✅ Hoàn tất!");
}
